import numpy as np
import pandas as pd

# ---- Cell 2 ----
X = np.linspace(1, 10, 30)

# ---- Cell 4 ----
def fit_analytical(x, target):
    x_mean = np.mean(x)
    y_mean = np.mean(target)
    numerator = np.sum((x - x_mean) * (target - y_mean))
    denominator = np.sum((x - x_mean) ** 2)
    slope = numerator / denominator
    intercept = y_mean - slope * x_mean
    return intercept, slope

b0_analytical, b1_analytical = fit_analytical(X, y)
y_pred_analytical = b0_analytical + b1_analytical * X

print(f'Intercept (b0): {b0_analytical:.4f}')
print(f'Slope (b1):     {b1_analytical:.4f}')
print(f'Equation: Score = {b0_analytical:.2f} + {b1_analytical:.2f} × Study Hours')

# ---- Cell 5 ----
plt.figure(figsize=(8, 5))
plt.scatter(X, y, label='Actual data', color='royalblue', edgecolor='black')
plt.plot(X, y_pred_analytical, label='Best-fit line', color='crimson', linewidth=2.5)
plt.xlabel('Study Hours')
plt.ylabel('Exam Score')
plt.title('Analytical Linear Regression')
plt.legend()
plt.show()

# ---- Cell 6 ----
def fit_gradient_descent(x, target, learning_rate=0.01, iterations=3000):
    intercept, slope = 0.0, 0.0
    n = len(x)
    cost_history = []

    for _ in range(iterations):
        predictions = intercept + slope * x
        errors = target - predictions
        cost_history.append(np.mean(errors ** 2))

        intercept_gradient = (-2 / n) * np.sum(errors)
        slope_gradient = (-2 / n) * np.sum(x * errors)
        intercept -= learning_rate * intercept_gradient
        slope -= learning_rate * slope_gradient

    return intercept, slope, np.array(cost_history)

b0_gd, b1_gd, costs = fit_gradient_descent(X, y)
y_pred_gd = b0_gd + b1_gd * X

print(f'Gradient-descent intercept: {b0_gd:.4f}')
print(f'Gradient-descent slope:     {b1_gd:.4f}')
print(f'Final MSE:                  {costs[-1]:.4f}')

# ---- Cell 7 ----
fig, axes = plt.subplots(1, 2, figsize=(13, 4.5))
axes[0].plot(costs, color='darkorange')
axes[0].set_title('Convergence of Gradient Descent')
axes[0].set_xlabel('Iteration')
axes[0].set_ylabel('Mean Squared Error')

axes[1].scatter(X, y, color='royalblue', edgecolor='black', label='Actual data')
axes[1].plot(X, y_pred_gd, color='green', linewidth=2.5, label='Gradient-descent line')
axes[1].set_title('Fitted Regression Line')
axes[1].set_xlabel('Study Hours')
axes[1].set_ylabel('Exam Score')
axes[1].legend()
plt.tight_layout()
plt.show()

# ---- Cell 8 ----
def regression_metrics(actual, predicted):
    residuals = actual - predicted
    mae = np.mean(np.abs(residuals))
    mse = np.mean(residuals ** 2)
    rmse = np.sqrt(mse)
    ss_res = np.sum(residuals ** 2)
    ss_tot = np.sum((actual - np.mean(actual)) ** 2)
    r2 = 1 - ss_res / ss_tot
    return {'MAE': mae, 'MSE': mse, 'RMSE': rmse, 'R²': r2}

results = pd.DataFrame({
    'Analytical': regression_metrics(y, y_pred_analytical),
    'Gradient Descent': regression_metrics(y, y_pred_gd)
}).T
display(results.round(4))

# ---- Cell 2 ----
def sigmoid(z):
    # Clipping avoids overflow when z is very large in magnitude.
    z = np.clip(z, -500, 500)
    return 1 / (1 + np.exp(-z))

z_values = np.linspace(-8, 8, 400)

plt.plot(z_values, sigmoid(z_values), linewidth=2.5)

# ---- Cell 5 ----
class LogisticRegressionScratch:
    def __init__(self, learning_rate=0.05, epochs=2500):
        self.learning_rate = learning_rate
        self.epochs = epochs
        self.weights = None
        self.loss_history = []

    def _add_intercept(self, X):
        return np.c_[np.ones(X.shape[0]), X]

    def fit(self, X, y):
        Xb = self._add_intercept(X)
        y = np.asarray(y).reshape(-1)
        self.weights = np.zeros(Xb.shape[1])
        for _ in range(self.epochs):
            probabilities = sigmoid(Xb @ self.weights)
            probabilities = np.clip(probabilities, 1e-12, 1 - 1e-12)
            loss = -np.mean(y * np.log(probabilities) + (1-y) * np.log(1-probabilities))
            gradient = (Xb.T @ (probabilities - y)) / len(y)
            self.weights -= self.learning_rate * gradient
            self.loss_history.append(loss)
        return self

    def predict_proba(self, X):
        Xb = self._add_intercept(X)
        p1 = sigmoid(Xb @ self.weights)
        return np.c_[1-p1, p1]

    def predict(self, X, threshold=0.5):
        return (self.predict_proba(X)[:, 1] >= threshold).astype(int)

scratch_model = LogisticRegressionScratch(learning_rate=0.05, epochs=2500)

def show_metrics(y_true, y_pred, y_score, title='Model evaluation'):

    display(pd.DataFrame(results, index=[title]).round(4))
    print(classification_report(y_true, y_pred, target_names=['benign (0)', 'malignant (1)'], zero_division=0))
    return results

# ---- Cell 4 ----
# Visualize each model's decision regions. The displayed axes use standardized coordinates.
def plot_decision_regions(ax, model, X, y, title):
    scaler = model.named_steps["standardscaler"]
    svc = model.named_steps["svc"]
    X_scaled = scaler.transform(X)
    pad = 0.6
    x0_min, x0_max = X_scaled[:, 0].min() - pad, X_scaled[:, 0].max() + pad
    x1_min, x1_max = X_scaled[:, 1].min() - pad, X_scaled[:, 1].max() + pad
    xx, yy = np.meshgrid(np.linspace(x0_min, x0_max, 250), np.linspace(x1_min, x1_max, 250))
    grid = np.c_[xx.ravel(), yy.ravel()]
    zz = svc.predict(grid).reshape(xx.shape)
    ax.contourf(xx, yy, zz, alpha=0.22, cmap="coolwarm")
    ax.scatter(X_scaled[:, 0], X_scaled[:, 1], c=y, cmap="coolwarm", s=18, edgecolor="k", linewidth=0.2)
    # Mark support vectors in standardized coordinates
    sv = svc.support_vectors_
    ax.scatter(sv[:, 0], sv[:, 1], s=75, facecolors="none", edgecolors="black", linewidths=1.0, label="Support vectors")
    ax.set_title(title); ax.set_xlabel("Standardized feature 1"); ax.set_ylabel("Standardized feature 2")
    ax.legend(loc="best", fontsize=8)

fig, axes = plt.subplots(1, 3, figsize=(16, 4.5))

print("Test examples:", X_test.shape[0])
print("Features:", iris.feature_names)

pipeline = Pipeline([

    ("classifier", LogisticRegression(max_iter=1000))

    param_grid=param_grid,

    cv=cv,
    n_jobs=-1,

)

# ---- Cell 1 ----
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

from sklearn.datasets import load_breast_cancer
from sklearn.model_selection import train_test_split, GridSearchCV, RandomizedSearchCV, StratifiedKFold
from sklearn.pipeline import Pipeline
from sklearn.preprocessing import StandardScaler
from sklearn.svm import SVC
from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score, confusion_matrix, classification_report
from scipy.stats import loguniform

RANDOM_STATE = 42

cancer = load_breast_cancer(as_frame=True)
X = cancer.data
y = cancer.target

print("Feature matrix:", X.shape)
print("Class names:", list(cancer.target_names))
print("Class counts:")
print(y.value_counts().rename(index=dict(enumerate(cancer.target_names))))
X.head()

# ---- Cell 2 ----
X_train, X_test, y_train, y_test = train_test_split(
    X, y, test_size=0.20, random_state=RANDOM_STATE, stratify=y
)
print(f"Training samples: {len(X_train)}")
print(f"Test samples: {len(X_test)}")

# ---- Cell 3 ----
baseline = Pipeline([
    ("scaler", StandardScaler()),
    ("svc", SVC())
])
baseline.fit(X_train, y_train)
baseline_pred = baseline.predict(X_test)

def metric_row(name, y_true, y_pred):
    return {
        "Model": name,
        "Accuracy": accuracy_score(y_true, y_pred),
        "Precision": precision_score(y_true, y_pred),
        "Recall": recall_score(y_true, y_pred),
        "F1": f1_score(y_true, y_pred),
    }

results = [metric_row("Baseline SVC", y_test, baseline_pred)]
pd.DataFrame(results).set_index("Model").round(4)

# ---- Cell 4 ----
cv = StratifiedKFold(n_splits=5, shuffle=True, random_state=RANDOM_STATE)
grid_params = {
    "svc__C": [0.1, 1, 10, 100],
    "svc__gamma": ["scale", 0.001, 0.01, 0.1],
    "svc__kernel": ["rbf", "linear"]
}

grid_search = GridSearchCV(
    estimator=Pipeline([("scaler", StandardScaler()), ("svc", SVC())]),
    param_grid=grid_params,
    scoring="f1",
    cv=cv,
    n_jobs=-1,
    refit=True,
    return_train_score=True
)
grid_search.fit(X_train, y_train)
print("Best CV F1:", round(grid_search.best_score_, 4))
print("Best parameters:", grid_search.best_params_)

# ---- Cell 5 ----
random_params = {
    "svc__C": loguniform(1e-2, 1e3),
    "svc__gamma": loguniform(1e-5, 1e0),
    "svc__kernel": ["rbf", "linear"]
}

random_search = RandomizedSearchCV(
    estimator=Pipeline([("scaler", StandardScaler()), ("svc", SVC())]),
    param_distributions=random_params,
    n_iter=30,
    scoring="f1",
    cv=cv,
    n_jobs=-1,
    random_state=RANDOM_STATE,
    refit=True,
    return_train_score=True
)
random_search.fit(X_train, y_train)
print("Best CV F1:", round(random_search.best_score_, 4))
print("Best parameters:", random_search.best_params_)

# ---- Cell 6 ----
grid_pred = grid_search.predict(X_test)
random_pred = random_search.predict(X_test)

results = [
    metric_row("Baseline SVC", y_test, baseline_pred),
    metric_row("GridSearchCV SVC", y_test, grid_pred),
    metric_row("RandomizedSearchCV SVC", y_test, random_pred),
]
comparison = pd.DataFrame(results).set_index("Model")
comparison.round(4)

# ---- Cell 7 ----
ax = comparison[["Accuracy", "Precision", "Recall", "F1"]].plot(
    kind="bar", figsize=(10, 5), ylim=(0.85, 1.0), rot=0
)
ax.set_ylabel("Test-set score")
ax.set_title("Baseline and tuned SVC performance")
ax.grid(axis="y", alpha=0.3)
plt.legend(loc="lower right")
plt.tight_layout()
plt.show()

# ---- Cell 8 ----
best_model = grid_search.best_estimator_
best_pred = best_model.predict(X_test)

print("Classification report (GridSearchCV model):")
print(classification_report(y_test, best_pred, target_names=cancer.target_names))
print("Confusion matrix:")
print(confusion_matrix(y_test, best_pred))

# ---- Cell 9 ----
cv_results = pd.DataFrame(grid_search.cv_results_)
cols = ["rank_test_score", "mean_test_score", "std_test_score", "mean_train_score", "param_svc__C", "param_svc__gamma", "param_svc__kernel"]
cv_results[cols].sort_values("rank_test_score").head(10).reset_index(drop=True).round(4)
