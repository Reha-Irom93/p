# EXPERIMENT 1: Perceptron (NumPy) with decision boundary
#####################################################################
import numpy as np
import matplotlib.pyplot as plt
from sklearn.datasets import make_blobs

X, y = make_blobs(n_samples=100, centers=2, cluster_std=1.0, random_state=6)

class Perceptron:
    def __init__(self, lr=0.1, epochs=20):
        self.lr, self.epochs = lr, epochs
    def fit(self, X, y):
        self.w = np.zeros(X.shape[1]); self.b = 0.0; self.errors = []
        for _ in range(self.epochs):
            err = 0
            for xi, ti in zip(X, y):
                yi = self.predict(xi)
                upd = self.lr * (ti - yi)
                self.w += upd * xi
                self.b += upd
                err += int(upd != 0)
            self.errors.append(err)
        return self
    def predict(self, X):
        return np.where(np.dot(X, self.w) + self.b >= 0, 1, 0)

p = Perceptron().fit(X, y)
print("Accuracy:", np.mean(p.predict(X) == y))

xx = np.linspace(X[:, 0].min() - 1, X[:, 0].max() + 1, 100)
yy = -(p.w[0] * xx + p.b) / p.w[1]
plt.scatter(X[:, 0], X[:, 1], c=y, cmap="bwr")
plt.plot(xx, yy, "k-"); plt.title("Perceptron decision boundary"); plt.show()
plt.plot(p.errors, marker="o"); plt.xlabel("Epoch"); plt.ylabel("Misclassifications"); plt.show()

# EXPERIMENT 3: Backpropagation from scratch (XOR, one hidden layer)
#####################################################################
import numpy as np
import matplotlib.pyplot as plt
np.random.seed(1)

X = np.array([[0,0],[0,1],[1,0],[1,1]], dtype=float)
Y = np.array([[0],[1],[1],[0]], dtype=float)

sigmoid = lambda z: 1 / (1 + np.exp(-z))
W1 = np.random.randn(2, 4); b1 = np.zeros((1, 4))
W2 = np.random.randn(4, 1); b2 = np.zeros((1, 1))
lr = 0.5
loss_hist, g1_hist, g2_hist = [], [], []

for epoch in range(10000):
    # forward
    A1 = sigmoid(X @ W1 + b1)
    A2 = sigmoid(A1 @ W2 + b2)
    loss = np.mean((A2 - Y) ** 2)
    # backward
    dZ2 = (A2 - Y) * A2 * (1 - A2)
    dW2 = A1.T @ dZ2;   db2 = dZ2.sum(0, keepdims=True)
    dZ1 = (dZ2 @ W2.T) * A1 * (1 - A1)
    dW1 = X.T @ dZ1;    db1 = dZ1.sum(0, keepdims=True)
    # update
    W2 -= lr * dW2; b2 -= lr * db2
    W1 -= lr * dW1; b1 -= lr * db1
    loss_hist.append(loss)
    g1_hist.append(np.linalg.norm(dW1)); g2_hist.append(np.linalg.norm(dW2))

print("Predictions:", np.round(A2.ravel(), 3))
plt.subplot(1, 2, 1); plt.plot(loss_hist); plt.title("Loss")
plt.subplot(1, 2, 2); plt.plot(g1_hist, label="|dW1|"); plt.plot(g2_hist, label="|dW2|")
plt.title("Gradient norms"); plt.legend(); plt.show()

# EXPERIMENT 4: Compare activation functions
#####################################################################
import matplotlib.pyplot as plt
from sklearn.datasets import make_moons
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
import tensorflow as tf
from tensorflow.keras import layers, models

X, y = make_moons(n_samples=2000, noise=0.2, random_state=0)
Xtr, Xte, ytr, yte = train_test_split(X, y, test_size=0.2, random_state=0)
sc = StandardScaler().fit(Xtr); Xtr, Xte = sc.transform(Xtr), sc.transform(Xte)

def build(act):
    a = (lambda: layers.LeakyReLU(0.1)) if act == "leaky_relu" else (lambda: layers.Activation(act))
    m = models.Sequential([layers.Input(shape=(2,))])
    for _ in range(3):
        m.add(layers.Dense(32)); m.add(a())
    m.add(layers.Dense(1, activation="sigmoid"))
    m.compile(optimizer="adam", loss="binary_crossentropy", metrics=["accuracy"])
    return m

for act in ["sigmoid", "tanh", "relu", "leaky_relu"]:
    tf.keras.utils.set_random_seed(0)
    h = build(act).fit(Xtr, ytr, epochs=50, batch_size=32, validation_data=(Xte, yte), verbose=0)
    plt.plot(h.history["loss"], label=act)
    print(act, "final val acc:", round(h.history["val_accuracy"][-1], 4))
plt.xlabel("Epoch"); plt.ylabel("Training loss"); plt.legend(); plt.show()

# EXPERIMENT 5: Linear Reg, Logistic Reg, k-NN from scratch
#####################################################################
import numpy as np, time
from sklearn.datasets import make_regression, make_classification
from sklearn.model_selection import train_test_split

class LinearReg:
    def fit(self, X, y, lr=0.01, n=1000):
        self.w = np.zeros(X.shape[1]); self.b = 0
        for _ in range(n):
            e = X @ self.w + self.b - y
            self.w -= lr * X.T @ e / len(y); self.b -= lr * e.mean()
        return self
    def predict(self, X): return X @ self.w + self.b

class LogisticReg:
    def fit(self, X, y, lr=0.1, n=1000):
        self.w = np.zeros(X.shape[1]); self.b = 0
        for _ in range(n):
            p = 1 / (1 + np.exp(-(X @ self.w + self.b)))
            self.w -= lr * X.T @ (p - y) / len(y); self.b -= lr * (p - y).mean()
        return self
    def predict(self, X): return (1 / (1 + np.exp(-(X @ self.w + self.b))) >= 0.5).astype(int)

class KNN:
    def __init__(self, k=5): self.k = k
    def fit(self, X, y): self.X, self.y = X, y; return self
    def predict(self, X):
        out = []
        for x in X:
            d = np.sqrt(((self.X - x) ** 2).sum(1))
            out.append(np.bincount(self.y[np.argsort(d)[:self.k]]).argmax())
        return np.array(out)

# Regression
Xr, yr = make_regression(n_samples=500, n_features=3, noise=10, random_state=1)
Xa, Xb, ya, yb = train_test_split(Xr, yr, random_state=1)
t = time.time(); lr_m = LinearReg().fit(Xa, ya); tl = time.time() - t
pred = lr_m.predict(Xb)
r2 = 1 - ((yb - pred) ** 2).sum() / ((yb - yb.mean()) ** 2).sum()
print("Linear Regression R2:", round(r2, 3), "time:", round(tl, 4))

# Classification
Xc, yc = make_classification(n_samples=500, n_features=4, n_informative=3, n_redundant=0, random_state=1)
Xa, Xb, ya, yb = train_test_split(Xc, yc, random_state=1)
for name, m in [("Logistic Regression", LogisticReg()), ("kNN", KNN(5))]:
    t = time.time(); m.fit(Xa, ya); pr = m.predict(Xb)
    print(name, "accuracy:", np.mean(pr == yb), "time:", round(time.time() - t, 4))

# EXPERIMENT 6: k-Means and Hierarchical clustering

import numpy as np, matplotlib.pyplot as plt
from sklearn.datasets import load_iris
from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans, AgglomerativeClustering
from sklearn.decomposition import PCA
from sklearn.metrics import silhouette_score
from scipy.cluster.hierarchy import dendrogram, linkage

X = StandardScaler().fit_transform(load_iris().data)
X2 = PCA(2).fit_transform(X)

# Elbow and silhouette for k
for k in range(2, 7):
    km = KMeans(k, n_init=10, random_state=0).fit(X)
    print(k, "inertia:", round(km.inertia_, 1), "silhouette:", round(silhouette_score(X, km.labels_), 3))

km = KMeans(3, n_init=10, random_state=0).fit(X)
ag = AgglomerativeClustering(3, linkage="ward").fit(X)
print("KMeans silhouette:", silhouette_score(X, km.labels_))
print("Hierarchical silhouette:", silhouette_score(X, ag.labels_))

plt.subplot(1, 2, 1); plt.scatter(X2[:, 0], X2[:, 1], c=km.labels_); plt.title("K-Means")
plt.subplot(1, 2, 2); plt.scatter(X2[:, 0], X2[:, 1], c=ag.labels_); plt.title("Hierarchical"); plt.show()
dendrogram(linkage(X, "ward")); plt.title("Dendrogram"); plt.show()

# EXPERIMENT 7: FNN vs CNN (CIFAR-10) and RNN (IMDB)

import time
import tensorflow as tf
from tensorflow.keras import layers, models

# ---- CIFAR-10: FNN and CNN ----
(Xtr, ytr), (Xte, yte) = tf.keras.datasets.cifar10.load_data()
Xtr, Xte = Xtr / 255.0, Xte / 255.0

fnn = models.Sequential([layers.Input((32, 32, 3)), layers.Flatten(),
    layers.Dense(512, activation="relu"), layers.Dense(256, activation="relu"),
    layers.Dense(10, activation="softmax")])
cnn = models.Sequential([layers.Input((32, 32, 3)),
    layers.Conv2D(32, 3, activation="relu"), layers.MaxPooling2D(),
    layers.Conv2D(64, 3, activation="relu"), layers.MaxPooling2D(),
    layers.Flatten(), layers.Dense(64, activation="relu"),
    layers.Dense(10, activation="softmax")])

for name, m in [("FNN", fnn), ("CNN", cnn)]:
    m.compile("adam", "sparse_categorical_crossentropy", metrics=["accuracy"])
    t = time.time(); m.fit(Xtr, ytr, epochs=5, batch_size=64, verbose=0)
    print(name, "acc:", m.evaluate(Xte, yte, verbose=0)[1], "time:", round(time.time() - t, 1), "params:", m.count_params())

# ---- IMDB: RNN ----
(Xa, ya), (Xb, yb) = tf.keras.datasets.imdb.load_data(num_words=10000)
Xa = tf.keras.utils.pad_sequences(Xa, maxlen=200); Xb = tf.keras.utils.pad_sequences(Xb, maxlen=200)

    layers.SimpleRNN(32), layers.Dense(1, activation="sigmoid")])

t = time.time(); rnn.fit(Xa, ya, epochs=3, batch_size=64, validation_split=0.2, verbose=0)
print("RNN acc:", rnn.evaluate(Xb, yb, verbose=0)[1], "time:", round(time.time() - t, 1))

# EXPERIMENT 8: Gradient descent for linear regression
#####################################################################
import numpy as np
import matplotlib.pyplot as plt
np.random.seed(0)

X = 2 * np.random.rand(100, 1)
y = 4 + 3 * X + np.random.randn(100, 1)      # true: b=4, w=3

w, b, lr, iters = 0.0, 0.0, 0.1, 200
m = len(X); costs = []
for i in range(iters):
    pred = w * X + b
    err = pred - y
    costs.append((err ** 2).sum() / (2 * m))
    w -= lr * (err * X).sum() / m
    b -= lr * err.sum() / m

print("Learned w=%.3f, b=%.3f" % (w, b))
plt.subplot(1, 2, 1); plt.plot(costs); plt.xlabel("Iteration"); plt.ylabel("Cost J"); plt.title("Convergence")
plt.subplot(1, 2, 2); plt.scatter(X, y); plt.plot(X, w * X + b, "r"); plt.title("Fitted line"); plt.show()

import numpy as np
import matplotlib.pyplot as plt
np.random.seed(0)

A = np.array([[1, 2], [3, 4]]); B = np.array([[5, 6], [7, 8]])
print("Add:\n", A + B)
print("Matrix multiply:\n", A @ B)
print("Element-wise:\n", A * B)
print("Transpose:\n", A.T)

# Network 3 -> 4 -> 1, sigmoid, MSE
X = np.random.rand(5, 3); Y = np.random.randint(0, 2, (5, 1)).astype(float)
W1 = np.random.randn(3, 4) * 0.5; b1 = np.zeros((1, 4))
W2 = np.random.randn(4, 1) * 0.5; b2 = np.zeros((1, 1))
sig = lambda z: 1 / (1 + np.exp(-z))

lr, losses = 0.5, []
for i in range(500):
    Z1 = X @ W1 + b1; A1 = sig(Z1)
    Z2 = A1 @ W2 + b2; A2 = sig(Z2)
    losses.append(np.mean((A2 - Y) ** 2))
    dZ2 = (A2 - Y) * A2 * (1 - A2) / len(X)
    dW2 = A1.T @ dZ2; db2 = dZ2.sum(0, keepdims=True)
    dZ1 = (dZ2 @ W2.T) * A1 * (1 - A1)
    dW1 = X.T @ dZ1; db1 = dZ1.sum(0, keepdims=True)
    W2 -= lr * dW2; b2 -= lr * db2; W1 -= lr * dW1; b1 -= lr * db1
    if i == 0:
        print("Hidden activations (1st iter):\n", np.round(A1, 3))
        print("dW1 (1st iter):\n", np.round(dW1, 4))

plt.plot(losses); plt.title("Loss"); plt.xlabel("Iteration"); plt.show()
plt.imshow(A1, cmap="viridis"); plt.colorbar(); plt.title("Hidden layer outputs"); plt.show()

# EXPERIMENT 10: CNN variants on MNIST
#####################################################################
import tensorflow as tf
from tensorflow.keras import layers, models

(Xtr, ytr), (Xte, yte) = tf.keras.datasets.mnist.load_data()
Xtr = (Xtr / 255.0)[..., None]; Xte = (Xte / 255.0)[..., None]

def make_cnn(n_conv=2, filters=32, ksize=3, pool="max"):
    P = layers.MaxPooling2D if pool == "max" else layers.AveragePooling2D
    m = models.Sequential([layers.Input((28, 28, 1))])
    for i in range(n_conv):
        m.add(layers.Conv2D(filters * 2 ** i, ksize, activation="relu", padding="same"))
        m.add(P())
    m.add(layers.Flatten()); m.add(layers.Dropout(0.3))
    m.add(layers.Dense(64, activation="relu")); m.add(layers.Dense(10, activation="softmax"))
    m.compile("adam", "sparse_categorical_crossentropy", metrics=["accuracy"])
    return m

configs = [dict(n_conv=1, ksize=3), dict(n_conv=2, ksize=3),
           dict(n_conv=2, ksize=5), dict(n_conv=2, ksize=3, pool="avg")]
for cfg in configs:
    m = make_cnn(**cfg)
    m.fit(Xtr, ytr, epochs=3, batch_size=128, validation_split=0.1, verbose=0)
    print(cfg, "test acc:", round(m.evaluate(Xte, yte, verbose=0)[1], 4), "params:", m.count_params())

# EXPERIMENT 11: Transfer learning (MobileNetV2) vs scratch
#####################################################################
import pathlib, tensorflow as tf
from tensorflow.keras import layers, models

url = "https://storage.googleapis.com/download.tensorflow.org/example_images/flower_photos.tgz"
data_dir = pathlib.Path(tf.keras.utils.get_file("flower_photos", origin=url, untar=True))
IMG = (160, 160)
train = tf.keras.utils.image_dataset_from_directory(data_dir, validation_split=0.2, subset="training", seed=1, image_size=IMG, batch_size=32)
val = tf.keras.utils.image_dataset_from_directory(data_dir, validation_split=0.2, subset="validation", seed=1, image_size=IMG, batch_size=32)
n_cls = len(train.class_names)

# A) Feature extraction
base = tf.keras.applications.MobileNetV2(input_shape=IMG + (3,), include_top=False, weights="imagenet")
base.trainable = False
inp = layers.Input(IMG + (3,))
x = tf.keras.applications.mobilenet_v2.preprocess_input(inp)
x = base(x, training=False)
x = layers.GlobalAveragePooling2D()(x); x = layers.Dropout(0.2)(x)
out = layers.Dense(n_cls, activation="softmax")(x)
tl = models.Model(inp, out)
tl.compile(tf.keras.optimizers.Adam(1e-3), "sparse_categorical_crossentropy", metrics=["accuracy"])
tl.fit(train, validation_data=val, epochs=5)

# B) Fine-tuning: unfreeze top layers, low LR
base.trainable = True
for l in base.layers[:-30]: l.trainable = False
tl.compile(tf.keras.optimizers.Adam(1e-5), "sparse_categorical_crossentropy", metrics=["accuracy"])
tl.fit(train, validation_data=val, epochs=5)

# C) From scratch for comparison
scratch = models.Sequential([layers.Input(IMG + (3,)), layers.Rescaling(1. / 255),
    layers.Conv2D(32, 3, activation="relu"), layers.MaxPooling2D(),
    layers.Conv2D(64, 3, activation="relu"), layers.MaxPooling2D(),
    layers.Flatten(), layers.Dense(128, activation="relu"), layers.Dense(n_cls, activation="softmax")])
scratch.compile("adam", "sparse_categorical_crossentropy", metrics=["accuracy"])
scratch.fit(train, validation_data=val, epochs=5)

# EXPERIMENT 12: RNN, LSTM, GRU on IMDB sentiment
#####################################################################
import time
import tensorflow as tf
from tensorflow.keras import layers, models

(Xtr, ytr), (Xte, yte) = tf.keras.datasets.imdb.load_data(num_words=10000)
Xtr = tf.keras.utils.pad_sequences(Xtr, maxlen=200)
Xte = tf.keras.utils.pad_sequences(Xte, maxlen=200)

def build(cell):
    return models.Sequential([layers.Input((200,)), layers.Embedding(10000, 64),
        cell(64), layers.Dense(1, activation="sigmoid")])

for name, cell in [("SimpleRNN", layers.SimpleRNN), ("LSTM", layers.LSTM), ("GRU", layers.GRU)]:
    m = build(cell)
    m.compile("adam", "binary_crossentropy", metrics=["accuracy"])
    t = time.time()
    m.fit(Xtr, ytr, epochs=3, batch_size=64, validation_split=0.2, verbose=0)
    loss, acc = m.evaluate(Xte, yte, verbose=0)
    print(f"{name}: test acc={acc:.4f}, loss={loss:.4f}, time={time.time() - t:.0f}s, params={m.count_params()}")
