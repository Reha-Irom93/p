# EXPERIMENT 1: Introduction to Scikit-learn for Supervised Learning
######################################################################

# ---- Cell 1 ----
import warnings  # imports warnings module
warnings.filterwarnings('ignore')  # hides warning messages to keep the output clean

import numpy as np  # imports NumPy: numerical arrays and math as np
import pandas as pd  # imports pandas: tables (DataFrame/Series) as pd
import matplotlib.pyplot as plt  # imports Matplotlib: plotting as plt
import seaborn as sns  # imports Seaborn: statistical plots as sns
import sklearn  # imports Scikit-learn library

from sklearn.datasets import load_iris  # imports from sklearn.datasets: load_iris = loads the built-in Iris dataset
from sklearn.model_selection import train_test_split, cross_val_score, GridSearchCV  # imports from sklearn.model_selection: train_test_split = splits data into training and test sets; cross_val_score = computes cross-validation scores; GridSearchCV = exhaustive hyperparameter search with cross-validation
from sklearn.preprocessing import StandardScaler  # imports from sklearn.preprocessing: StandardScaler = standardizes features (zero mean, unit variance)
from sklearn.pipeline import Pipeline  # imports from sklearn.pipeline: Pipeline = chains preprocessing and model steps
from sklearn.linear_model import LogisticRegression  # imports from sklearn.linear_model: LogisticRegression = Logistic Regression classifier
from sklearn.neighbors import KNeighborsClassifier  # imports from sklearn.neighbors: KNeighborsClassifier = K-Nearest Neighbours classifier
from sklearn.tree import DecisionTreeClassifier, plot_tree  # imports from sklearn.tree: DecisionTreeClassifier = Decision Tree classifier; plot_tree = draws a decision tree
from sklearn.svm import SVC  # imports from sklearn.svm: SVC = Support Vector Classifier
from sklearn.metrics import (  # imports the evaluation metrics listed on the next lines
    accuracy_score, confusion_matrix, classification_report,  # imports: accuracy_score = accuracy metric; confusion_matrix = confusion matrix; classification_report = per-class precision/recall/F1 report
    precision_score, recall_score, f1_score  # imports: precision_score = precision metric; recall_score = recall metric; f1_score = F1-score metric
)  # closes the bracket opened above

print('Scikit-learn version:', sklearn.__version__)  # prints 'Scikit-learn version:' with the value

# ---- Cell 2 ----
iris = load_iris(as_frame=True)  # loads Iris with features/target as pandas objects
X = iris.data  # feature table (4 measurements per flower)
y = iris.target  # target labels (flower species 0, 1, 2)

df = X.copy()  # copies the features so the original stays unchanged
df['target'] = y  # adds the numeric class label as a column
df['species'] = y.map(dict(enumerate(iris.target_names)))  # adds the species name by mapping 0/1/2 to names

print('Feature names:', list(X.columns))  # prints 'Feature names:' with the value
print('Target names:', list(iris.target_names))  # prints 'Target names:' with the value
print('Dataset shape:', df.shape)  # prints 'Dataset shape:' with the value
display(df.head())  # shows the result as a table: df.head()

# ---- Cell 3 ----
print('Data types and missing values:')  # prints the text 'Data types and missing values:'
display(pd.DataFrame({'dtype': X.dtypes, 'missing': X.isna().sum()}))  # shows the result as a table: pd.DataFrame({'dtype': X.dtypes, 'missing': X.isna

print('\nDescriptive statistics:')  # prints the text 'Descriptive statistics:'
display(X.describe().round(2))  # shows the result as a table: X.describe().round(2)

print('\nClass distribution:')  # prints the text 'Class distribution:'
display(df['species'].value_counts())  # shows the result as a table: df['species'].value_counts()

# ---- Cell 4 ----
plt.figure(figsize=(8, 5))  # creates a new figure with the given size (inches)
sns.scatterplot(data=df, x='petal length (cm)', y='petal width (cm)',  # scatter plot of petal length vs width coloured by species
                hue='species', style='species', s=90)  # colour by this column (hue='species'); marker style by column (style='species'); marker size (s=90))
plt.title('Iris flowers by petal measurements')  # sets the plot title: Iris flowers by petal measurements
plt.grid(alpha=0.2)  # adds light grid lines
plt.show()  # renders the figure

# ---- Cell 5 ----
X_train, X_test, y_train, y_test = train_test_split(  # splits data into train and test parts (arguments on the next lines)
    X, y, test_size=0.20, random_state=42, stratify=y  # features, target, 20% test, fixed seed, class proportions kept (stratify)
)  # closes the bracket opened above

print('Training samples:', X_train.shape[0])  # prints 'Training samples:' with the value
print('Testing samples:', X_test.shape[0])  # prints 'Testing samples:' with the value
print('Training class counts:', y_train.value_counts().sort_index().to_dict())  # prints 'Training class counts:' with the value
print('Testing class counts:', y_test.value_counts().sort_index().to_dict())  # prints 'Testing class counts:' with the value

# ---- Cell 6 ----
logistic_model = Pipeline([  # pipeline: scaler then logistic regression
    ('scaler', StandardScaler()),  # step 1: standardize features
    ('classifier', LogisticRegression(max_iter=1000, random_state=42))  # step 2: logistic regression classifier
])  # closes the bracket opened above

logistic_model.fit(X_train, y_train)  # trains the pipeline on training data
y_pred_lr = logistic_model.predict(X_test)  # predicts classes for the test samples

print('Accuracy:', round(accuracy_score(y_test, y_pred_lr), 4))  # prints 'Accuracy:' with the value
print('\nClassification report:')  # prints the text 'Classification report:'
print(classification_report(y_test, y_pred_lr, target_names=iris.target_names))  # prints the per-class report (arguments follow)

# ---- Cell 7 ----
cm = confusion_matrix(y_test, y_pred_lr)  # confusion matrix of actual vs predicted
plt.figure(figsize=(6, 4))  # creates a new figure with the given size (inches)
sns.heatmap(cm, annot=True, fmt='d', cmap='Blues',  # draws the matrix as a colour heatmap (options on the following lines)
            xticklabels=iris.target_names, yticklabels=iris.target_names)  # x-axis tick labels (xticklabels=iris.target_names); y-axis tick labels (yticklabels=iris.target_names))
plt.xlabel('Predicted class')  # labels the x-axis: Predicted class
plt.ylabel('Actual class')  # labels the y-axis: Actual class
plt.title('Logistic Regression — Confusion Matrix')  # sets the plot title: Logistic Regression — Confusion Matrix
plt.show()  # renders the figure

# ---- Cell 8 ----
models = {  # dictionary of models to compare (name: model)
    'Logistic Regression': Pipeline([  # model 1: scaled logistic regression
        ('scaler', StandardScaler()),  # step 1: standardize features
        ('classifier', LogisticRegression(max_iter=1000, random_state=42))  # step 2: logistic regression classifier
    ]),  # closes this pipeline entry
    'KNN': Pipeline([  # model 2: scaled KNN
        ('scaler', StandardScaler()),  # step 1: standardize features
        ('classifier', KNeighborsClassifier(n_neighbors=5))  # step 2: KNN with k = 5
    ]),  # closes this pipeline entry
    'Decision Tree': DecisionTreeClassifier(max_depth=3, random_state=42),  # model 3: decision tree, depth limited to 3
    'SVM (RBF)': Pipeline([  # model 4: scaled SVM with RBF kernel
        ('scaler', StandardScaler()),  # step 1: standardize features
        ('classifier', SVC(kernel='rbf', probability=True, random_state=42))  # step 2: SVM with RBF kernel (probability estimates on)
    ])  # closes the bracket opened above
}  # closes the bracket opened above

results = []  # empty list to collect results
predictions = {}  # empty dictionary to store each model's predictions

for name, model in models.items():  # loop: repeats the block for each name, model in models.items()
    model.fit(X_train, y_train)  # trains the model on the training data
    pred = model.predict(X_test)  # predicts the test labels
    predictions[name] = pred  # stores this model's predictions by name
    results.append({  # adds one row of results to the list
        'Model': name,  # model name column
        'Accuracy': accuracy_score(y_test, pred),  # accuracy of this model
        'Macro Precision': precision_score(y_test, pred, average='macro'),  # macro-averaged precision (equal weight per class)
        'Macro Recall': recall_score(y_test, pred, average='macro'),  # macro-averaged recall
        'Macro F1': f1_score(y_test, pred, average='macro')  # macro-averaged F1-score
    })  # closes the bracket opened above

results_df = pd.DataFrame(results).sort_values('Macro F1', ascending=False)  # results as a table, best Macro F1 first
display(results_df.round(4))  # shows the result as a table: results_df.round(4)

# ---- Cell 9 ----
plot_df = results_df.melt(id_vars='Model', var_name='Metric', value_name='Score')  # converts the table to long format for plotting
plt.figure(figsize=(10, 5))  # creates a new figure with the given size (inches)
sns.barplot(data=plot_df, x='Model', y='Score', hue='Metric')  # grouped bar chart of the metrics per model
plt.ylim(0.80, 1.02)  # sets the y-axis range to 0.80-1.02 to see small differences
plt.title('Test-set performance comparison')  # sets the plot title: Test-set performance comparison
plt.xticks(rotation=15)  # tilts the x labels by 15 degrees
plt.legend(bbox_to_anchor=(1.02, 1), loc='upper left')  # draws the legend
plt.tight_layout()  # adjusts spacing so labels do not overlap
plt.show()  # renders the figure

# ---- Cell 10 ----
cv_results = []  # empty list for cross-validation results
for name, model in models.items():  # loop: repeats the block for each name, model in models.items()
    scores = cross_val_score(model, X_train, y_train, cv=5, scoring='accuracy')  # 5-fold cross-validation accuracy scores on the training data
    cv_results.append({  # adds one row of cross-validation results
        'Model': name,  # model name column
        'Fold scores': np.round(scores, 3),  # accuracy of each of the 5 folds, rounded
        'Mean CV accuracy': scores.mean(),  # average accuracy over folds
        'Std. deviation': scores.std()  # spread of the fold scores (stability)
    })  # closes the bracket opened above

cv_df = pd.DataFrame(cv_results).sort_values('Mean CV accuracy', ascending=False)  # cross-validation table, best mean accuracy first
display(cv_df)  # shows the result as a table: cv_df

# ---- Cell 11 ----
knn_pipeline = Pipeline([  # pipeline: scaler then KNN
    ('scaler', StandardScaler()),  # step 1: standardize features
    ('classifier', KNeighborsClassifier())  # step 2: KNN (its settings are tuned by the grid search)
])  # closes the bracket opened above

parameter_grid = {  # hyperparameter values to try
    'classifier__n_neighbors': list(range(1, 16)),  # k from 1 to 15
    'classifier__weights': ['uniform', 'distance'],  # neighbour weighting: equal or by inverse distance
    'classifier__p': [1, 2]  # distance type: 1 = Manhattan, 2 = Euclidean (original note: 1 = Manhattan; 2 = Euclidean distance)
}  # closes the bracket opened above

grid_search = GridSearchCV(  # grid search over the settings above (arguments follow)
    estimator=knn_pipeline,  # model/pipeline to tune (estimator=knn_pipeline)
    param_grid=parameter_grid,  # parameter combinations to try (param_grid=parameter_grid)
    cv=5,  # cross-validation setting (folds / splitter) (cv=5)
    scoring='accuracy',  # metric used to rank candidates (scoring='accuracy')
    n_jobs=-1  # -1 = use all CPU cores (n_jobs=-1)
)  # closes the bracket opened above
grid_search.fit(X_train, y_train)  # runs the search: cross-validates every combination

print('Best parameters:', grid_search.best_params_)  # prints 'Best parameters:' with the value
print('Best cross-validation accuracy:', round(grid_search.best_score_, 4))  # prints 'Best cross-validation accuracy:' with the value

best_knn = grid_search.best_estimator_  # the best KNN model found, already refitted
y_pred_tuned = best_knn.predict(X_test)  # tuned model's predictions on the test set
print('Tuned KNN test accuracy:', round(accuracy_score(y_test, y_pred_tuned), 4))  # prints 'Tuned KNN test accuracy:' with the value

# ---- Cell 12 ----
tree_model = models['Decision Tree']  # picks the trained decision tree from the dictionary
plt.figure(figsize=(14, 7))  # creates a new figure with the given size (inches)
plot_tree(tree_model, feature_names=X.columns, class_names=iris.target_names,  # draws the tree: split rules, class names, colours (options follow)
          filled=True, rounded=True, fontsize=9)  # colour nodes by class (filled=True); rounded node boxes (rounded=True); font size (fontsize=9))
plt.title('Decision Tree learned from the training data')  # sets the plot title: Decision Tree learned from the training data
plt.show()  # renders the figure

# ---- Cell 13 ----
new_flowers = pd.DataFrame(  # table of 3 new unseen flowers
    [[5.1, 3.5, 1.4, 0.2],  # flower 1: sepal length, sepal width, petal length, petal width (cm)
     [6.0, 2.9, 4.5, 1.5],  # flower 2 measurements
     [6.7, 3.1, 5.6, 2.4]],  # flower 3 measurements
    columns=X.columns  # use the training feature names
)  # closes the bracket opened above

predicted_ids = best_knn.predict(new_flowers)  # predicted class numbers for the new flowers
probabilities = best_knn.predict_proba(new_flowers)  # probability of each class for each flower

prediction_table = new_flowers.copy()  # copy of the table to add predictions to
prediction_table['predicted_species'] = iris.target_names[predicted_ids]  # adds the predicted species names
for index, species in enumerate(iris.target_names):  # loop: repeats the block for each index, species in enumerate(iris.target_names)
    prediction_table[f'P({species})'] = probabilities[:, index]  # adds a column with the probability of this species

display(prediction_table.round(3))  # shows the result as a table: prediction_table.round(3)

# EXPERIMENT 2: Exploring Unsupervised Learning with K-Means Clustering
######################################################################

# ---- Cell 1 ----
# Import required libraries
import numpy as np  # imports NumPy: numerical arrays and math as np
import pandas as pd  # imports pandas: tables (DataFrame/Series) as pd
import matplotlib.pyplot as plt  # imports Matplotlib: plotting as plt
import seaborn as sns  # imports Seaborn: statistical plots as sns

from sklearn.cluster import KMeans  # imports from sklearn.cluster: KMeans = K-Means clustering algorithm
from sklearn.datasets import make_blobs  # imports from sklearn.datasets: make_blobs = generates clustered (blob) data
from sklearn.metrics import silhouette_score, silhouette_samples  # imports from sklearn.metrics: silhouette_score = average silhouette score; silhouette_samples = per-sample silhouette values
from sklearn.preprocessing import StandardScaler  # imports from sklearn.preprocessing: StandardScaler = standardizes features (zero mean, unit variance)

RANDOM_STATE = 42  # seed constant for reproducible results
np.random.seed(RANDOM_STATE)  # fixes NumPy's random generator
sns.set_theme(style='whitegrid', context='notebook')  # applies a Seaborn plot theme
print('Libraries imported successfully.')  # prints the text 'Libraries imported successfully.'

# ---- Cell 2 ----
centers = [(28, 25), (32, 78), (68, 30), (72, 75)]  # centres of the 4 hidden customer groups
cluster_std = [6.5, 7.0, 7.5, 6.0]  # spread of each group
X_raw, _ = make_blobs(  # generates 2-D customer data (true labels ignored with _)
    n_samples=320, centers=centers, cluster_std=cluster_std,  # number of samples to generate (n_samples=320); cluster centres (centers=centers); spread of each cluster (cluster_std=cluster_std)
    random_state=RANDOM_STATE  # fixed seed so results are reproducible (random_state=RANDOM_STATE)
)  # closes the bracket opened above

df = pd.DataFrame(X_raw, columns=['Annual Income (₹ lakh scaled)', 'Spending Score'])  # puts the data in a table with income and spending columns
df['Annual Income (₹ lakh scaled)'] = df['Annual Income (₹ lakh scaled)'].clip(10, 100)  # limits income values to 10-100
df['Spending Score'] = df['Spending Score'].clip(1, 100)  # limits spending score to 1-100

print('Dataset shape:', df.shape)  # prints 'Dataset shape:' with the value
display(df.head(10))  # shows the result as a table: df.head(10)

# ---- Cell 3 ----
df.info()  # prints column types and non-null counts
print('\nMissing values:')  # prints the text 'Missing values:'
display(df.isna().sum().to_frame('Missing'))  # shows the result as a table: df.isna().sum().to_frame('Missing')
print('Summary statistics:')  # prints the text 'Summary statistics:'
display(df.describe().round(2))  # shows the result as a table: df.describe().round(2)

# ---- Cell 4 ----
fig, axes = plt.subplots(1, 3, figsize=(17, 4.5))  # creates 3 side-by-side plots
sns.histplot(df.iloc[:, 0], kde=True, ax=axes[0], color='steelblue')  # histogram of income with density curve (first plot)
axes[0].set_title('Income Distribution')  # sets the plot title: Income Distribution
sns.histplot(df.iloc[:, 1], kde=True, ax=axes[1], color='darkorange')  # histogram of spending score with density curve (second plot)
axes[1].set_title('Spending Score Distribution')  # sets the plot title: Spending Score Distribution
sns.scatterplot(data=df, x=df.columns[0], y=df.columns[1], ax=axes[2], s=50)  # scatter of income vs spending (third plot)
axes[2].set_title('Customers Before Clustering')  # sets the plot title: Customers Before Clustering
plt.tight_layout()  # adjusts spacing so labels do not overlap
plt.show()  # renders the figure

# ---- Cell 5 ----
features = df.columns.tolist()  # list of feature column names
scaler = StandardScaler()  # creates the scaler
X_scaled = scaler.fit_transform(df[features])  # learns mean/std and standardizes the features
scaled_df = pd.DataFrame(X_scaled, columns=features)  # scaled data as a table
display(scaled_df.describe().round(3))  # shows the result as a table: scaled_df.describe().round(3)

# ---- Cell 6 ----
k_values = range(2, 11)  # tries K = 2 to 10
inertias, silhouette_scores = [], []  # empty lists for the two measures

for k in k_values:  # loop: repeats the block for each k in k_values
    model = KMeans(n_clusters=k, init='k-means++', n_init=20, random_state=RANDOM_STATE)  # K-Means with k clusters, k-means++ start, 20 restarts
    labels = model.fit_predict(X_scaled)  # trains K-Means and returns each point's cluster label
    inertias.append(model.inertia_)  # stores the within-cluster sum of squares
    silhouette_scores.append(silhouette_score(X_scaled, labels))  # stores the silhouette score of this clustering

evaluation = pd.DataFrame({  # table of K, inertia and silhouette
    'K': list(k_values), 'Inertia': inertias, 'Silhouette Score': silhouette_scores  # K, inertia and silhouette columns
})  # closes the bracket opened above
display(evaluation.round(4))  # shows the result as a table: evaluation.round(4)

# ---- Cell 7 ----
fig, axes = plt.subplots(1, 2, figsize=(13, 4.5))  # creates 2 side-by-side plots
axes[0].plot(list(k_values), inertias, marker='o')  # elbow plot: inertia vs K
axes[0].set(title='Elbow Method', xlabel='Number of clusters (K)', ylabel='Inertia')  # title and axis labels of the elbow plot
axes[1].plot(list(k_values), silhouette_scores, marker='o', color='darkgreen')  # silhouette score vs K
axes[1].set(title='Silhouette Analysis', xlabel='Number of clusters (K)', ylabel='Silhouette score')  # title and axis labels of the silhouette plot
axes[1].set_ylim(0, 1)  # y-axis from 0 to 1
plt.tight_layout()  # adjusts spacing so labels do not overlap
plt.show()  # renders the figure

best_k_by_silhouette = int(evaluation.loc[evaluation['Silhouette Score'].idxmax(), 'K'])  # K with the highest silhouette score
print('Best K by silhouette score:', best_k_by_silhouette)  # prints 'Best K by silhouette score:' with the value

# ---- Cell 8 ----
FINAL_K = 4  # chosen number of clusters
kmeans = KMeans(  # final K-Means model (settings on the next lines)
    n_clusters=FINAL_K, init='k-means++', n_init=20,  # number of clusters K (n_clusters=FINAL_K); centroid initialization method (init='k-means++'); number of restarts; best run is kept (n_init=20)
    max_iter=300, random_state=RANDOM_STATE  # maximum number of solver iterations (max_iter=300); fixed seed so results are reproducible (random_state=RANDOM_STATE)
)  # closes the bracket opened above
df['Cluster'] = kmeans.fit_predict(X_scaled)  # trains K-Means and stores each customer's cluster

print('Iterations used:', kmeans.n_iter_)  # prints 'Iterations used:' with the value
print('Final inertia:', round(kmeans.inertia_, 3))  # prints 'Final inertia:' with the value
print('Silhouette score:', round(silhouette_score(X_scaled, df['Cluster']), 3))  # prints 'Silhouette score:' with the value
display(df.head())  # shows the result as a table: df.head()

# ---- Cell 9 ----
centroids_original = scaler.inverse_transform(kmeans.cluster_centers_)  # converts centroids back to original units
centroids_df = pd.DataFrame(centroids_original, columns=features)  # centroids as a table
centroids_df.index.name = 'Cluster'  # names the row index 'Cluster'
display(centroids_df.round(2))  # shows the result as a table: centroids_df.round(2)

plt.figure(figsize=(9, 6))  # creates a new figure with the given size (inches)
sns.scatterplot(  # scatter of customers coloured by cluster (options follow)
    data=df, x=features[0], y=features[1], hue='Cluster',  # data source (data=df); x-axis column (x=features[0]); y-axis column (y=features[1]); colour by this column (hue='Cluster')
    palette='tab10', s=65, alpha=0.75  # colour palette (palette='tab10'); marker size (s=65); transparency (alpha=0.75)
)  # closes the bracket opened above
plt.scatter(  # draws the centroids as black X markers (options follow)
    centroids_original[:, 0], centroids_original[:, 1],  # x and y of the centroids
    c='black', marker='X', s=260, label='Centroids', edgecolors='white'  # black X markers, size 260, white edge, legend label 'Centroids'
)  # closes the bracket opened above
plt.title('K-Means Customer Segments')  # sets the plot title: K-Means Customer Segments
plt.legend(title='Cluster', bbox_to_anchor=(1.02, 1), loc='upper left')  # draws the legend
plt.tight_layout()  # adjusts spacing so labels do not overlap
plt.show()  # renders the figure

# ---- Cell 10 ----
cluster_profile = df.groupby('Cluster').agg(  # summary per cluster (below)
    Customers=('Cluster', 'size'),  # number of customers in the cluster
    Mean_Income=(features[0], 'mean'),  # average income of the cluster
    Mean_Spending=(features[1], 'mean')  # average spending score of the cluster
).round(2)  # rounds to 2 decimals
cluster_profile['Percentage'] = (100 * cluster_profile['Customers'] / len(df)).round(1)  # share of customers in each cluster (%)
display(cluster_profile)  # shows the result as a table: cluster_profile

income_mid = df[features[0]].median()  # median income, used as the high/low cut-off
spending_mid = df[features[1]].median()  # median spending, used as the high/low cut-off

def describe_segment(row):  # defines function/method describe_segment with parameters (row)
    income = 'High-income' if row['Mean_Income'] >= income_mid else 'Lower-income'  # labels the cluster high or lower income
    spending = 'high-spending' if row['Mean_Spending'] >= spending_mid else 'low-spending'  # labels the cluster high or low spending
    return f'{income}, {spending}'  # returns the combined segment name

cluster_profile['Suggested Segment'] = cluster_profile.apply(describe_segment, axis=1)  # adds a descriptive segment name per cluster
display(cluster_profile[['Suggested Segment']])  # shows the result as a table: cluster_profile[['Suggested Segment']]

# ---- Cell 11 ----
df['Silhouette'] = silhouette_samples(X_scaled, df['Cluster'])  # silhouette value of every customer
display(df.groupby('Cluster')['Silhouette'].agg(['mean', 'min', 'max']).round(3))  # shows the result as a table: df.groupby('Cluster')['Silhouette'].agg(['mean', '

plt.figure(figsize=(9, 4.5))  # creates a new figure with the given size (inches)
sns.boxplot(data=df, x='Cluster', y='Silhouette', hue='Cluster', palette='tab10', legend=False)  # box plot of silhouette values per cluster
plt.axhline(0, color='red', linestyle='--', linewidth=1)  # red dashed line at 0 (negative = poorly fitted points)
plt.title('Silhouette Values by Cluster')  # sets the plot title: Silhouette Values by Cluster
plt.tight_layout()  # adjusts spacing so labels do not overlap
plt.show()  # renders the figure

# ---- Cell 12 ----
new_customers = pd.DataFrame({  # table of 4 new customers
    features[0]: [25, 75, 30, 70],  # incomes of the new customers
    features[1]: [20, 80, 85, 25]  # spending scores of the new customers
})  # closes the bracket opened above
new_scaled = scaler.transform(new_customers)  # scales new data with the already fitted scaler
new_customers['Predicted Cluster'] = kmeans.predict(new_scaled)  # assigns each new customer to the nearest centroid
new_customers['Suggested Segment'] = new_customers['Predicted Cluster'].map(  # looks up the segment name of each predicted cluster
    cluster_profile['Suggested Segment']  # adds a descriptive segment name per cluster
)  # closes the bracket opened above
display(new_customers)  # shows the result as a table: new_customers

# EXPERIMENT 3: Implementing Linear Regression from Scratch
######################################################################

# ---- Cell 1 ----
import numpy as np  # imports NumPy: numerical arrays and math as np
import pandas as pd  # imports pandas: tables (DataFrame/Series) as pd
import matplotlib.pyplot as plt  # imports Matplotlib: plotting as plt

np.random.seed(42)  # fixes the random seed
plt.style.use('seaborn-v0_8-whitegrid')  # applies a plotting style

# ---- Cell 2 ----
X = np.linspace(1, 10, 30)  # 30 evenly spaced study hours from 1 to 10
noise = np.random.normal(0, 4, size=X.shape)  # random noise (mean 0, std 4) for each point
y = 32 + 6.2 * X + noise  # true relation score = 32 + 6.2 * hours plus noise

# ---- Cell 4 ----
def fit_analytical(x, target):  # defines function/method fit_analytical with parameters (x, target)
    x_mean = np.mean(x)  # mean of x
    y_mean = np.mean(target)  # mean of y
    numerator = np.sum((x - x_mean) * (target - y_mean))  # sum of (x - x mean)(y - y mean)
    denominator = np.sum((x - x_mean) ** 2)  # sum of (x - x mean) squared
    slope = numerator / denominator  # slope b1 by least squares
    intercept = y_mean - slope * x_mean  # intercept b0 = y mean - b1 * x mean
    return intercept, slope  # returns the fitted b0 and b1

def fit_gradient_descent(x, target, learning_rate=0.01, iterations=3000):  # defines function/method fit_gradient_descent with parameters (x, target, learning_rate=0.01, iterations=3000)
    intercept, slope = 0.0, 0.0  # starts both parameters at 0
    n = len(x)  # number of observations
    cost_history = []  # list to record the cost at each iteration

    for _ in range(iterations):  # loop: repeats the block for each _ in range(iterations)
        predictions = intercept + slope * x  # current predictions
        errors = target - predictions  # residuals (actual - predicted)
        cost_history.append(np.mean(errors ** 2))  # records the MSE of this iteration

        intercept_gradient = (-2 / n) * np.sum(errors)  # gradient of MSE with respect to intercept
        slope_gradient = (-2 / n) * np.sum(x * errors)  # gradient of MSE with respect to slope
        intercept -= learning_rate * intercept_gradient  # updates intercept against the gradient
        slope -= learning_rate * slope_gradient  # updates slope against the gradient

    return intercept, slope, np.array(cost_history)  # returns parameters and the cost history

b0_gd, b1_gd, costs = fit_gradient_descent(X, y)  # runs gradient descent (default rate 0.01, 3000 iterations)

def regression_metrics(actual, predicted):  # defines function/method regression_metrics with parameters (actual, predicted)
    residuals = actual - predicted  # residuals = actual - predicted
    mae = np.mean(np.abs(residuals))  # MAE: mean absolute error
    mse = np.mean(residuals ** 2)  # MSE: mean squared error
    rmse = np.sqrt(mse)  # RMSE: square root of MSE
    ss_res = np.sum(residuals ** 2)  # residual sum of squares
    ss_tot = np.sum((actual - np.mean(actual)) ** 2)  # total sum of squares around the mean
    r2 = 1 - ss_res / ss_tot  # R² = 1 - SS_res/SS_tot
    return {'MAE': mae, 'MSE': mse, 'RMSE': rmse, 'R²': r2}  # returns all four metrics as a dictionary

class LogisticRegressionScratch:  # defines the class LogisticRegressionScratch
    def __init__(self, learning_rate=0.05, epochs=2500):  # defines function/method __init__ with parameters (self, learning_rate=0.05, epochs=2500)
        self.learning_rate = learning_rate  # stores the learning rate
        self.epochs = epochs  # stores the number of iterations
        self.weights = None  # weights are set during fit
        self.loss_history = []  # list to record the loss each epoch

    def _add_intercept(self, X):  # defines function/method _add_intercept with parameters (self, X)
        return np.c_[np.ones(X.shape[0]), X]  # adds a column of ones for the intercept

    def fit(self, X, y):  # defines function/method fit with parameters (self, X, y)
        Xb = self._add_intercept(X)  # feature matrix with intercept column
        y = np.asarray(y).reshape(-1)  # target as a flat array
        self.weights = np.zeros(Xb.shape[1])  # initial weights = 0
        for _ in range(self.epochs):  # loop: repeats the block for each _ in range(self.epochs)
            probabilities = sigmoid(Xb @ self.weights)  # predicted probabilities
            probabilities = np.clip(probabilities, 1e-12, 1 - 1e-12)  # keeps probabilities away from 0 and 1 to avoid log(0)
            loss = -np.mean(y * np.log(probabilities) + (1-y) * np.log(1-probabilities))  # binary cross-entropy loss
            gradient = (Xb.T @ (probabilities - y)) / len(y)  # gradient (1/n) X^T (p - y)
            self.weights -= self.learning_rate * gradient  # gradient-descent weight update
            self.loss_history.append(loss)  # records the loss
        return self  # returns the trained model

    def predict_proba(self, X):  # defines function/method predict_proba with parameters (self, X)
        Xb = self._add_intercept(X)  # feature matrix with intercept column
        p1 = sigmoid(Xb @ self.weights)  # probability of class 1
        return np.c_[1-p1, p1]  # probabilities of class 0 and class 1

    def predict(self, X, threshold=0.5):  # defines function/method predict with parameters (self, X, threshold=0.5)
        return (self.predict_proba(X)[:, 1] >= threshold).astype(int)  # class 1 if probability >= threshold, else 0

# Visualize each model's decision regions. The displayed axes use standardized coordinates.
def plot_decision_regions(ax, model, X, y, title):  # defines function/method plot_decision_regions with parameters (ax, model, X, y, title)
    scaler = model.named_steps["standardscaler"]  # gets the fitted scaler
    svc = model.named_steps["svc"]  # gets the fitted SVM
    X_scaled = scaler.transform(X)  # scales the points to plot
    pad = 0.6  # extra margin around the plotted area
    x0_min, x0_max = X_scaled[:, 0].min() - pad, X_scaled[:, 0].max() + pad  # x-range of the plot with padding
    x1_min, x1_max = X_scaled[:, 1].min() - pad, X_scaled[:, 1].max() + pad  # y-range of the plot with padding
    xx, yy = np.meshgrid(np.linspace(x0_min, x0_max, 250), np.linspace(x1_min, x1_max, 250))  # grid of points covering the plot area
    grid = np.c_[xx.ravel(), yy.ravel()]  # grid points as a list of (x, y) pairs
    zz = svc.predict(grid).reshape(xx.shape)  # predicts the class at every grid point
    ax.contourf(xx, yy, zz, alpha=0.22, cmap="coolwarm")  # fills the regions by predicted class
    ax.scatter(X_scaled[:, 0], X_scaled[:, 1], c=y, cmap="coolwarm", s=18, edgecolor="k", linewidth=0.2)  # plots the data points coloured by class
    # Mark support vectors in standardized coordinates
    sv = svc.support_vectors_  # coordinates of the support vectors
    ax.scatter(sv[:, 0], sv[:, 1], s=75, facecolors="none", edgecolors="black", linewidths=1.0, label="Support vectors")  # circles the support vectors
    ax.set_title(title); ax.set_xlabel("Standardized feature 1"); ax.set_ylabel("Standardized feature 2")  # sets the plot title:  | labels the x-axis: Standardized feature 1 | labels the y-axis: Standardized feature 2
    ax.legend(loc="best", fontsize=8)  # draws the legend

# ---- Cell 1 ----
from sklearn.datasets import load_iris  # imports from sklearn.datasets: load_iris = loads the built-in Iris dataset
from sklearn.model_selection import train_test_split, StratifiedKFold, GridSearchCV  # imports from sklearn.model_selection: train_test_split = splits data into training and test sets; StratifiedKFold = k-fold splitter that keeps class proportions; GridSearchCV = exhaustive hyperparameter search with cross-validation
from sklearn.pipeline import Pipeline  # imports from sklearn.pipeline: Pipeline = chains preprocessing and model steps
from sklearn.preprocessing import StandardScaler  # imports from sklearn.preprocessing: StandardScaler = standardizes features (zero mean, unit variance)
from sklearn.linear_model import LogisticRegression  # imports from sklearn.linear_model: LogisticRegression = Logistic Regression classifier
from sklearn.svm import SVC  # imports from sklearn.svm: SVC = Support Vector Classifier
from sklearn.ensemble import RandomForestClassifier  # imports from sklearn.ensemble: RandomForestClassifier = Random Forest classifier
from sklearn.metrics import accuracy_score, classification_report  # imports from sklearn.metrics: accuracy_score = accuracy metric; classification_report = per-class precision/recall/F1 report
import pandas as pd  # imports pandas: tables (DataFrame/Series) as pd

# ---- Cell 2 ----
iris = load_iris()  # loads the Iris dataset (NumPy arrays)
X, y = iris.data, iris.target  # features and target arrays

X_train, X_test, y_train, y_test = train_test_split(  # splits data into train and test parts (arguments on the next lines)
    X, y,  # features and target to split (remaining arguments on the following lines)
    test_size=0.20,  # fraction of data used for testing (test_size=0.20)
    random_state=42,  # fixed seed so results are reproducible (random_state=42)
    stratify=y  # keep class proportions in both splits (stratify=y)
)  # closes the bracket opened above

print("Training examples:", X_train.shape[0])  # prints 'Training examples:' with the value
print("Test examples:", X_test.shape[0])  # prints 'Test examples:' with the value
print("Features:", iris.feature_names)  # prints 'Features:' with the value
print("Classes:", list(iris.target_names))  # prints 'Classes:' with the value

# ---- Cell 3 ----
pipeline = Pipeline([  # base pipeline: scaler then classifier
    ("scale", StandardScaler()),  # step 1: standardize
    ("classifier", LogisticRegression(max_iter=1000))  # step 2: placeholder classifier (replaced by the grid)
])  # closes the bracket opened above

param_grid = [  # list of parameter grids, one per candidate model
    {  # starts one parameter grid
        "classifier": [LogisticRegression(max_iter=1000)],  # candidate 1: logistic regression
        "classifier__C": [0.1, 1, 10]  # C values to try
    },  # closes the bracket opened above
    {  # starts one parameter grid
        "classifier": [SVC()],  # candidate 2: SVC
        "classifier__C": [0.1, 1, 10],  # C values to try
        "classifier__kernel": ["linear", "rbf"]  # kernels to try
    },  # closes the bracket opened above
    {  # starts one parameter grid
        "classifier": [RandomForestClassifier(random_state=42)],  # candidate 3: random forest
        "classifier__n_estimators": [100, 200],  # numbers of trees
        "classifier__max_depth": [None, 3, 5]  # depth limits
    }  # closes the bracket opened above
]  # closes the bracket opened above

cv = StratifiedKFold(n_splits=5, shuffle=True, random_state=42)  # 5 shuffled folds that keep class proportions
search = GridSearchCV(  # grid search over all candidates (arguments follow)
    estimator=pipeline,  # model/pipeline to tune (estimator=pipeline)
    param_grid=param_grid,  # parameter combinations to try (param_grid=param_grid)
    scoring="accuracy",  # metric used to rank candidates (scoring="accuracy")
    cv=cv,  # cross-validation setting (folds / splitter) (cv=cv)
    n_jobs=-1,  # -1 = use all CPU cores (n_jobs=-1)
    return_train_score=True  # also record training scores (return_train_score=True)
)  # closes the bracket opened above

# ---- Cell 4 ----
search.fit(X_train, y_train)  # runs the search on training data

print("Best classifier:", search.best_estimator_.named_steps["classifier"])  # prints 'Best classifier:' with the value
print("Best parameters:", search.best_params_)  # prints 'Best parameters:' with the value
print(f"Best mean cross-validation accuracy: {search.best_score_:.3f}")  # prints 'Best mean cross-validation accuracy: {search.best_score_:.3f}' with the value

# ---- Cell 5 ----
results = pd.DataFrame(search.cv_results_)  # CV results as a table
results[["mean_test_score", "std_test_score", "mean_train_score", "params"]].sort_values(  # selects score columns
    by="mean_test_score", ascending=False  # best first
).head(10).reset_index(drop=True)  # top 10 rows with a clean index

# ---- Cell 6 ----
y_pred = search.predict(X_test)  # best model's test predictions
print(f"Test accuracy: {accuracy_score(y_test, y_pred):.3f}")  # prints 'Test accuracy: {accuracy_score(y_test, y_pred):.3f}' with the value
print("\nClassification report:")  # prints the text 'Classification report:'
print(classification_report(y_test, y_pred, target_names=iris.target_names))  # prints the per-class report (arguments follow)

# ---- Cell 1 ----
import numpy as np  # imports NumPy: numerical arrays and math as np
import pandas as pd  # imports pandas: tables (DataFrame/Series) as pd
import matplotlib.pyplot as plt  # imports Matplotlib: plotting as plt

from sklearn.datasets import load_breast_cancer  # imports from sklearn.datasets: load_breast_cancer = loads the built-in Breast Cancer Wisconsin dataset
from sklearn.model_selection import train_test_split, GridSearchCV, RandomizedSearchCV, StratifiedKFold  # imports from sklearn.model_selection: train_test_split = splits data into training and test sets; GridSearchCV = exhaustive hyperparameter search with cross-validation; RandomizedSearchCV = random hyperparameter search with cross-validation; StratifiedKFold = k-fold splitter that keeps class proportions
from sklearn.pipeline import Pipeline  # imports from sklearn.pipeline: Pipeline = chains preprocessing and model steps
from sklearn.preprocessing import StandardScaler  # imports from sklearn.preprocessing: StandardScaler = standardizes features (zero mean, unit variance)
from sklearn.svm import SVC  # imports from sklearn.svm: SVC = Support Vector Classifier
from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score, confusion_matrix, classification_report  # imports from sklearn.metrics: accuracy_score = accuracy metric; precision_score = precision metric; recall_score = recall metric; f1_score = F1-score metric; confusion_matrix = confusion matrix; classification_report = per-class precision/recall/F1 report
from scipy.stats import loguniform  # imports from scipy.stats: loguniform = log-uniform distribution for random search

RANDOM_STATE = 42  # seed constant for reproducible results

cancer = load_breast_cancer(as_frame=True)  # loads the breast cancer data
X = cancer.data  # feature table
y = cancer.target  # target labels (0 = malignant, 1 = benign)

print("Feature matrix:", X.shape)  # prints 'Feature matrix:' with the value
print("Class names:", list(cancer.target_names))  # prints 'Class names:' with the value
print("Class counts:")  # prints the text 'Class counts:'
print(y.value_counts().rename(index=dict(enumerate(cancer.target_names))))  # prints the value
X.head()  # shows the first 5 rows

# ---- Cell 2 ----
X_train, X_test, y_train, y_test = train_test_split(  # splits data into train and test parts (arguments on the next lines)
    X, y, test_size=0.20, random_state=RANDOM_STATE, stratify=y  # features, target, 20% test, fixed seed, class proportions kept
)  # closes the bracket opened above
print(f"Training samples: {len(X_train)}")  # prints 'Training samples: {len(X_train)}' with the value
print(f"Test samples: {len(X_test)}")  # prints 'Test samples: {len(X_test)}' with the value

# ---- Cell 3 ----
baseline = Pipeline([  # baseline pipeline: scaler + default SVC
    ("scaler", StandardScaler()),  # step 1: standardize
    ("svc", SVC())  # step 2: SVC with default settings
])  # closes the bracket opened above
baseline.fit(X_train, y_train)  # trains the baseline
baseline_pred = baseline.predict(X_test)  # baseline test predictions

def metric_row(name, y_true, y_pred):  # defines function/method metric_row with parameters (name, y_true, y_pred)
    return {  # returns metrics as a dictionary
        "Model": name,  # model name
        "Accuracy": accuracy_score(y_true, y_pred),  # accuracy
        "Precision": precision_score(y_true, y_pred),  # precision
        "Recall": recall_score(y_true, y_pred),  # recall
        "F1": f1_score(y_true, y_pred),  # F1
    }  # closes the bracket opened above

results = [metric_row("Baseline SVC", y_test, baseline_pred)]  # first row: baseline metrics
pd.DataFrame(results).set_index("Model").round(4)  # metrics table indexed by model

# ---- Cell 4 ----
cv = StratifiedKFold(n_splits=5, shuffle=True, random_state=RANDOM_STATE)  # 5 shuffled folds that keep class proportions
grid_params = {  # hyperparameter grid
    "svc__C": [0.1, 1, 10, 100],  # C values to try
    "svc__gamma": ["scale", 0.001, 0.01, 0.1],  # gamma values to try
    "svc__kernel": ["rbf", "linear"]  # kernels to try
}  # closes the bracket opened above

grid_search = GridSearchCV(  # grid search over the settings above (arguments follow)
    estimator=Pipeline([("scaler", StandardScaler()), ("svc", SVC())]),  # pipeline to tune: scaler + SVC
    param_grid=grid_params,  # parameter combinations to try (param_grid=grid_params)
    scoring="f1",  # metric used to rank candidates (scoring="f1")
    cv=cv,  # cross-validation setting (folds / splitter) (cv=cv)
    n_jobs=-1,  # -1 = use all CPU cores (n_jobs=-1)
    refit=True,  # refit the best model on the whole training set (refit=True)
    return_train_score=True  # also record training scores (return_train_score=True)
)  # closes the bracket opened above
grid_search.fit(X_train, y_train)  # runs the search: cross-validates every combination
print("Best CV F1:", round(grid_search.best_score_, 4))  # prints 'Best CV F1:' with the value
print("Best parameters:", grid_search.best_params_)  # prints 'Best parameters:' with the value

# ---- Cell 5 ----
random_params = {  # distributions for random search
    "svc__C": loguniform(1e-2, 1e3),  # C sampled log-uniformly from 0.01 to 1000
    "svc__gamma": loguniform(1e-5, 1e0),  # gamma sampled log-uniformly from 1e-5 to 1
    "svc__kernel": ["rbf", "linear"]  # kernels to try
}  # closes the bracket opened above

random_search = RandomizedSearchCV(  # random search over 30 samples (arguments follow)
    estimator=Pipeline([("scaler", StandardScaler()), ("svc", SVC())]),  # pipeline to tune: scaler + SVC
    param_distributions=random_params,  # parameter distributions to sample (param_distributions=random_params)
    n_iter=30,  # number of random combinations (n_iter=30)
    scoring="f1",  # metric used to rank candidates (scoring="f1")
    cv=cv,  # cross-validation setting (folds / splitter) (cv=cv)
    n_jobs=-1,  # -1 = use all CPU cores (n_jobs=-1)
    random_state=RANDOM_STATE,  # fixed seed so results are reproducible (random_state=RANDOM_STATE)
    refit=True,  # refit the best model on the whole training set (refit=True)
    return_train_score=True  # also record training scores (return_train_score=True)
)  # closes the bracket opened above
random_search.fit(X_train, y_train)  # runs the random search
print("Best CV F1:", round(random_search.best_score_, 4))  # prints 'Best CV F1:' with the value
print("Best parameters:", random_search.best_params_)  # prints 'Best parameters:' with the value

# ---- Cell 6 ----
grid_pred = grid_search.predict(X_test)  # grid-search model's test predictions
random_pred = random_search.predict(X_test)  # random-search model's test predictions

results = [  # metrics rows of all three models
    metric_row("Baseline SVC", y_test, baseline_pred),  # baseline row
    metric_row("GridSearchCV SVC", y_test, grid_pred),  # grid-search row
    metric_row("RandomizedSearchCV SVC", y_test, random_pred),  # random-search row
]  # closes the bracket opened above
comparison = pd.DataFrame(results).set_index("Model")  # comparison table
comparison.round(4)  # shows the table with 4 decimals

# ---- Cell 7 ----
ax = comparison[["Accuracy", "Precision", "Recall", "F1"]].plot(  # bar chart of the four test metrics for each model (arguments follow)
    kind="bar", figsize=(10, 5), ylim=(0.85, 1.0), rot=0  # plot type (kind="bar"); figure size (inches) (figsize=(10, 5)); y-axis limits (ylim=(0.85, 1.0)); label rotation (rot=0)
)  # closes the bracket opened above
